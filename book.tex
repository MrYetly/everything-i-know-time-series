\chapter{Difference Equations}

This chapter is essentially a distillation of Hamilton's \cite{hamilton1994time} chapter of the same name.

The whole point of time series analysis is to uncover and describe the statistical properties and structure of a process across time. Certain processes may have a lot of structure, so much that advanced non-linear methods such as those enabled by machine learning are needed. Others have very little, and must be described by a very simple model as we are always worried about overfitting. So, we begin by studying the simplest of processes.

Assume we have the deterministic process

\begin{equation}
    y_t = \phi y_{t-1} + w_t \label{eq: first-order-deq}
\end{equation}

In order to find the state of this process at any time $t$, we need some kind of initial condition. We may take $y_{-1}$ as given, or if we want to look some arbitrary time setp $j$ into the future from a given time $t$, we may take $y_{t-1}$ as given. In either case, we can use recursion to find the state of $y_t$ at any future time $t$:

\begin{align*}
    y_t     &= \phi y_{t-1} + w_t, \\
    y_{t+1} &= \phi y_{t} + w_{t+1}     &&= \phi (\phi y_{t-1} + w_t) + w_{t+1}                     &&= \phi^2 y_{t-1} + \phi w_t + w_{t+1}, \\
    y_{t+2} &= \phi y_{t+1} + w_{t+2}   &&= \phi (\phi^2 y_{t-1} + \phi w_t + w_{t+1}) + w_{t+2}    &&= \phi^3 y_{t-1} + \phi^2 w_t + \phi w_{t+1} + w_{t+2}, \\
            &\vdots \\
    y_{t+j} &= \phi^{j+1} y_{t-1} + \sum_{k=0}^j \phi^{j-k} w_{t+k}. \refstepcounter{equation} \tag{\theequation} \label{eq:first-order-j-horizon}
\end{align*}

The last line shows the general form of this process at some future state: it has the classic characteristic of a time series process in that its state at any point in time depends on two things: an initial condition $y_{t-1}$ that encapsulates all of its history up until that point, and the shocks $w_{t+k}$ that have affected the process since the initial condition. The effect of a shock at time $t$ on the system at some future time $t+j$ is

\begin{equation}
\frac{\partial y_{t+j}}{\partial w_t} = \phi^j,
\end{equation}

\noindent and the effect of the initial condition is
\begin{equation}
\frac{\partial y_{t+j}}{\partial w_t} = \phi^{j+1}.
\end{equation}

\noindent So, it is clear that the stability of the process depends on $\phi$. If $-1 < \phi < 1$, the effect of the initial condition and all shocks at some future horizon $j$ goes to zero; if $1 < \phi < -1$, the effect of each compounds over time and the process explodes; if $\phi = \pm 1$, the shocks and the initial condition have a permanent effect of the process.

Now, instead of having $y_t$ depend on only its previous value and a contemporary shock, we'll have it depend on the previous $p$ values

\[
y_t = \phi_1 y_{t-1} + \phi_2 y_{t-2} + \cdots + \phi_p y_{t-p} + w_t.
\]

\noindent It is often more convenient to transform this into a first-order vector-valued equation,
 
\begin{equation}
\bm{\xi}_t = \bm{F} \bm{\xi}_{t-1} + \bm{v}_t \label{eq: p-order vector}
\end{equation}

\noindent where 

\[
\bm{\xi}_t = \begin{pmatrix}
y_t \\
y_{t-1} \\
\vdots \\
y_{t-p+1}
\end{pmatrix}
\]

\noindent is $p \times 1$,

\[
\bm{F} = \begin{pmatrix}
\phi_1  & \phi_2    & \cdots    & \phi_{p-1}    & \phi_p    \\
1       & 0         & \cdots    & 0         & 0         \\
0       & 1         & \cdots    & 0         & 0         \\
\vdots  & \vdots    & \cdots    & \vdots    & \vdots    \\
0       & 0         & \cdots    & 1         & 0
\end{pmatrix}
\]

\noindent is $p \times p$, and

\[
\bm{v}_t = \begin{pmatrix}
w_t \\
0 \\
\vdots \\
0
\end{pmatrix}.
\]

By a similar recursive operation before, if we have an initial condition at $\xi_{t-1}$ we can solve for the state of the process at some future horizon $j$,

\begin{align*}
\bm{\xi}_t &= \bm{F} \bm{\xi}_{t-1} + \bm{v}_t, \\
\bm{\xi}_{t+1} &= \bm{F} \bm{\xi}_{t} + \bm{v}_t &&= \bm{F} (\bm{F} \bm{\xi}_{t-1} + \bm{v}_t) + \bm{v}_{t+1} && = \bm{F}^2 \bm{\xi}_{t-1} + \bm{F} \bm{v}_t + \bm{v}_{t+1}, \\
\bm{\xi}_{t+2} &= \bm{F} \bm{\xi}_{t+1} + \bm{v}_{t+2} &&= \bm{F} (\bm{F}^2 \bm{\xi}_{t-1} + \bm{F} \bm{v}_t + \bm{v}_{t+1}) + \bm{v}_{t+2} && = \bm{F}^3 \bm{\xi}_{t-1} + \bm{F}^2 \bm{v}_t + \bm{F} \bm{v}_{t+1} + \bm{v}_{t+2}, \\
&\vdots \\
\bm{\xi}_{t+j} &= \bm{F}^{j+1} \bm{\xi}_{t-1} + \sum_{k=0}^j \bm{F}^{j-k} \bm{v}_{t-k} \refstepcounter{equation} \tag{\theequation} \label{eq:p-order-j-horizon}
\end{align*}

Note that the future value of interest $y_{t+j}$ is the first row of this last equation. If $f_{ij}^{n}$ is the $(i,j)$ element of $F^n$, then

\begin{align*}
y_{t+j} = &\underbrace{f_{11}^{j+1} y_{t-1} + f_{12}^{j+1} y_{t-2} + \cdots + f_{1p}^{j+1} y_{t-p}}_{\text{initial conditions}}  \\ 
& \underbrace{+ w_{t+j} + f_{11} w_{t+j-1} + \cdots + f_{11}^j w_t}_{\text{effect of shocks since initial state}}
\end{align*}

We can start to see how a process with a complicated structure can have its future values predicted, and how complicated the relationship between past and future values can be. For example, note how the effects of the shocks and the most recent period are all mediated by the $(1,1)$ element of $F^n$. 

\begin{align*}
f_{11}^1 &= \phi_1 \\
f_{11}^2 &= (\bm{F} \bm{F})_{1,1} = \left( \begin{pmatrix}
\phi_1  & \phi_2    & \cdots    & \phi_{p-1}    & \phi_p    \\
\vdots       & \vdots         & \vdots    & \vdots         & \vdots
\end{pmatrix} \begin{pmatrix}
\phi_1  & \cdots \\
1       & \cdots      \\
0       & \cdots      \\
\vdots  & \cdots \\
0       & \cdots      
\end{pmatrix} \right)_{1,1} = \phi_1^2 + \phi_2 \\
f_{11}^3 &= (\bm{F}^2 \bm{F})_{1,1} = \left( \begin{pmatrix}
\phi_1^2 + \phi_2  & \phi_1 \phi_2 + \phi_3    & \cdots    & \phi_1 \phi_{p-1} + \phi_p    & \phi_1 \phi_p    \\
\vdots       & \vdots         & \vdots    & \vdots         & \vdots
\end{pmatrix} \begin{pmatrix}
\phi_1  & \cdots \\
1       & \cdots      \\
0       & \cdots      \\
\vdots  & \cdots \\
0       & \cdots      
\end{pmatrix} \right)_{1,1} = \phi_1^3 + 2\phi_1\phi_2 + \phi_3 \\
&\vdots
\end{align*}


It is easy to see how the number of terms of $f_{11}^n$ blows up as $n$ increases. This is why it is often easier to reason with the vector-valued form of the system as shown in Equation \ref{eq:p-order-j-horizon}. If we want to get any kind of intuition for how the previous value we observe of a process may influence future values, it is worthwhile to work more with this vector-valued representation because it can provide analytical solutions for these influences.

If all of the eigenvalues of $\bm{F}$ are distinct, then we can write

\begin{align*}
\bm{F} = \bm{T} \bm{\Lambda} \bm{T}^{-1} \leftrightarrow \bm{F}^j = \bm{T} \bm{\Lambda}^j \bm{T}^{-1}
\end{align*}

\noindent where $\bm{\Lambda}$ is a diagonal matrix of eigenvalues of $\bm{F}$. After some algebraic manipulations, we find that

\[
\frac{\partial y_{t+j}}{\partial w_t} = c_1 \lambda_1^j + \cdots + c_p \lambda_p^j
\]

\noindent where $\lambda_k$ are the eigenvalues of $\bm{F}$ and $c_1 + \cdots + c_p = 1$. We can see that a similar exponential decay in the effect of the shock $w_t$ occurs for a p\textsuperscript{th}-order difference equations as for a first-order. Multiple lags do not get around the linearity of the system, which always leads to the effect of a shock either growing or decaying exponentially and monotonically over time.

\chapter{Lag Operators}

This chapter is essentially a distillation of Hamilton's \cite{hamilton1994time} chapter of the same name.

The lag operator $L$ is a convenient notational tool that allows difference equations and their properties to be expressed in elegant ways. For example, it allows us to write a first-order difference equation in the following equivalent ways,

\begin{align*}
y_t &= \phi y_{t-1} + w_t \\ 
&\updownarrow \\ 
y_t &= \phi L y_t + w_t \\ 
&\updownarrow \\ 
(1 - \phi L) y_t &= w_t \\
&\updownarrow \\ 
(1 + \phi L + \phi^2 L^2 + \cdots + \phi^t L^t)(1 - \phi L) y_t &= (1 + \phi L + \phi^2 L^2 + \cdots + \phi^t L^t)w_t \\
&\updownarrow \\ 
(1 - \phi^{t+1} L^{t+1}) y_t &= (1 + \phi L + \phi^2 L^2 + \cdots + \phi^t L^t)w_t
\end{align*}

There are two things to note about the last line here. First, that it is equivalent to Equation \ref{eq:first-order-j-horizon} with $t=0$ and $j=t$. A second, if $|\phi|<1$, as $t$ grows large, $(1 - \phi^{t+1} L^{t+1}) y_t \approx y_t$. So, we can define a new operator

\[
(1 - \phi L)^{-1} = \lim_{t \rightarrow \infty} (1 + \phi L + \cdots + \phi^t L^t)
\]

\noindent and write 

\[
(1 - \phi L) y_t = w_t \leftrightarrow y_t = (1 - \phi L)^{-1} w_t
\]

Note that it is only possible to express the process this way if $|\phi|<1$.

If we have a p-\textsuperscript{th} order difference equation, we can write 

\begin{align*}
y_t &= \phi_1 y_{t-1} + \cdots + \phi_p y_{t-p} + w_t \\
&\updownarrow \\
(1 - \phi_1 L - \cdots - \phi_p L^p) y_t &= w_t
&\updownarrow \\
(1 - \lambda_1 L) * \cdots * (1 - \lambda_p L) y_t = w_t
\end{align*}

Finding the $\lambda_k$ values is similar to factoring any other polynomial, and in fact it is the identical operation of find the eigenvalues of $\bm{F}$ in Equation \ref{eq:p-order-j-horizon}. Given this symmetry, if all of the eigenvalues found lie within the unit circle, then the system is stable, and we can write 

\[
y_t = (1 - \lambda_1 L)^{-1} * \cdots * (1 - \lambda_p L)^{-1} w_t
\]

\noindent After some involved algebra, we also find, unsurprisingly, that the effect of a shock $w_t$ at some horizon $j$ is the same as well.

\chapter{Stationary ARMA Processes}

This chapter is essentially a distillation of Hamilton's \cite{hamilton1994time} chapter of the same name.

A process $Y_t$ is \textbf{weakly stationary} if 

\begin{align*}
\mathbb{E}(Y_t) &= \mu && \forall t \\
\mathbb{E}[(Y_t - \mu)(Y_{t-j} - \mu)] &= \gamma_j && \forall t, j
\end{align*}

A process $Y_t$ is \textbf{strongly stationary} if for any values $j_1, ..., j_n$ the joint distribution of $(Y_t, Y_{t+j_1}, ..., Y_{t+j_n})$ only depends on the dates separating $j_1, ..., j_n$ and not on $t$.

A process $Y_t$ is \textbf{ergodic} for a certain characteristic (its mean, second moment, etc) if observing that time series over a long period of time allows you to make claims about that characteristic at a point in time. For example, for a time series, we never actually observe the processes repeatedly at any specific time. This would normally make it impossible to say what its mean is at any point in time. If the time series was ergodic for the mean, then we can say that a mean estimated from sufficient observations of a single time series across time is also a good estimate of the mean at any point in time.

\section{state space models - local level model}

two main assumptions are mentioned: the authors will assume normality or, when normality is not assumed, minimum variance linear unbiased estimation. I'm assuming the latter is BLUE. I have never thought that assuming normality and assuming BLUE were somehow solutions to the same problem. That's interesting.

A basic general time series model can be represented by 

\[
y_t = \mu_t + \gamma_t + \epsilon_t
\]

\begin{itemize}
    \item $y_t \in \{y_1, ..., y_n\}$, the observations
    \item $\mu_t$, trend (slowly varying, the state)
    \item $\gamma_t$, seasonal (periodic)
    \item $\epsilon_t$, error disturbance
\end{itemize}

In state-space time series estimation, we use the observations to make inferences about the state.

In general, when creating a state space model, we think about how to model $\mu_t$ and $\gamma_t$. A more specific, but still applicable model that illustrates this is the following random walk model known as a ``local level model''

\begin{align*}
y_t &= \alpha_t + \epsilon_t, &&\epsilon_t \sim N(0, \sigma_\epsilon^2), \\
\alpha_{t+1} &= \alpha_t + \eta_t, &&\eta_t \sim N(0, \sigma_\eta^2), \\
&   &&\epsilon_t \perp \eta_t.
\end{align*}

\noindent Here, the state is $\alpha_t$, and there is no seasonal component. The error component $\epsilon_t$ of $y_t$ differs from other representations of random walks in that it has a purely contemporaneous effect; it does not accumulate over time, the shocks $\eta_t$ to the state $\alpha_t$ do. It is as if we have a random walk with an extra disturnace term.

Additionally, we also assume $\alpha_1 \sim N(a_1, P_1)$, and that $a_1, P_1, \sigma_\epsilon^2, \sigma_\eta^2$ are known.

Since we are assuming a normal distribution for our fundamental variables and paramters, would could obtain closed form solutions for the moments of $y_t$ at any time $t$, but this is intractable as $n$ grows. This is why we use the technique known as filtering.

\subsection{filtering}

Assume

\begin{itemize}
    \item $Y_{t-1} = (y_1, ..., y_{t-1})'$ for $t=2, 3, ..., n$.
    \item $\alpha_t | Y_{t-1} \sim N(a_t, P_t)$, where $a_t, P_t$ are known. 
    \item $\alpha_t | Y_t \sim N(a_{t|t}, P_{t|t})$.
    \item $\alpha_{t+1} | Y_t \sim N(a_{t+1}, P_{t+1})$. 
\end{itemize}

The first thing that jumps out is that $\alpha_t$ is always unknown. We make guesses about them by estimating $a_t$ and $P_t$. We fill focus on two estimates involving the state of the system, a one-step ahead forecast $a_{t+1}$ and a ``filtered-estimator'' $a_{t|t}$ of the current state $\alpha_t$. Both estimates are made using information at time $t$, $Y_t$.

Filtering makes use of the one-step ahead prediction error of $y_t$, $v_t = y_t - a_t$. Some important moments of this prediction error are

\begin{align*}
\mathbb{E}(v_t|Y_{t-1}) &= \mathbb{E} (\alpha_t + \epsilon_t - a_t | Y_{t-1}) = a_t - a_t = 0, \\
Var(v_t|Y_{t-1}) &= Var (\alpha_t + \epsilon_t - a_t | Y_{t-1}) = P_t + \sigma_\epsilon^2, \\
\mathbb{E}(v_t|\alpha_t, Y_{t-1}) &= \mathbb{E} (\alpha_t + \epsilon_t - a_t | \alpha_t, Y_{t-1}) = \alpha_t - a_t, \\
Var(v_t| \alpha_t, Y_{t-1}) &= Var (\alpha_t + \epsilon_t - a_t | \alpha_t, Y_{t-1}) = \sigma_\epsilon^2, 
\end{align*}

\noindent Remember that at $t-1$, $a_t$ and $P_t$ are known, so $\mathbb{E}(a_t | Y_{t-1}) = a_t$ and $Var(a_t | Y_{t-1}) = 0$. We can update our expectations of the state with our prediction error by noticing that $p(\alpha_t | Y_t) = p(\alpha_t | Y_{t-1}, v_t)$ because if $Y_t$ is fixed, then so are both $(Y_{t-1}, y_t)$ and $(Y_{t-1}, v_t)$ because $a_t$ is known at time $t$ and $v_t = y_t - a_t$. Remembering that $p(\cdot)$ is a density function and all of our densities in this example are normal, 

\begin{align*}
p(\alpha_t | Y_{t-1}, v_t) &= p(\alpha_t, v_t | Y_{t-1})/p(v_t | Y_{t-1}), \\
&= p(\alpha_t | Y_{t-1}) p(v_t | \alpha_t, Y_{t-1}) /p(v_t | Y_{t-1}), \\
&= \text{constant} \times \text{exp}\left( \frac{-1}{2} Q \right).
\end{align*}

\noindent And remember that the normal density function is

\[
f(x) = \frac{1}{\sigma \sqrt{2\pi}} \text{exp}\left( \frac{-1}{2} \frac{(x-\mu)^2}{\sigma^2} \right)
\]

\noindent so whatever $Q$ is informs us of $\mu$ and $\sigma$ for the resulting normal distribution of interest, and thus defines it. $Q$ is built from the moments above

\begin{align*}
Q &= \frac{(\alpha_t - a_t)^2}{P_t} + \frac{(v_t - \alpha_t + a_t)^2}{\sigma_\epsilon^2} - \frac{v_t^2}{(P_t + \sigma_\epsilon^2)} \\
&= \frac{\alpha_t - \left(a_t +  \dfrac{P_t v_t}{P_t + \sigma_\epsilon^2} \right)}{ \dfrac{P_t \sigma_\epsilon^2}{P_t + \sigma_\epsilon^2}}
\end{align*}

\noindent so

\[
p(\alpha_t | Y_{t-1}, v_t) = p(\alpha_t | Y_t) = N\left( a_t + \frac{P_t v_t}{P_t + \sigma_\epsilon^2}, \frac{P_t \sigma_\epsilon^2}{P_t + \sigma_\epsilon^2} \right)
\]

\noindent and we defined $p(\alpha_t | Y_t) = N(a_{t|t}, P_{t|t})$ so

\begin{align*}
a_{t|t} &= a_t + \frac{P_t v_t}{P_t + \sigma_\epsilon^2}, \\
P_{t|t} &= \frac{P_t \sigma_\epsilon^2}{P_t + \sigma_\epsilon^2},
\end{align*}

\noindent are respectively the filter estimate of the current state $\alpha_t$, which is the most up-to-date expected value of the current state, and our most up-to-date quantification of how far away from this expectation the actual state may be (i.e. the most up to date variance of the state).

This up-to-date distribution of the current state is different from the unupdated verion, $p(\alpha_t | Y_{t-1}) = N(a_t, P_t)$. To understand this difference, we first need to understand the term $P_t + \sigma_\epsilon^2$. This is the total variance of our model given the information we have at time $t-1$, which is the same as the variance of our prediction error conditional on that same information,

\[
Var(y_t | Y_{t-1}) = Var(v_t | Y_{t-1}) = Var(\alpha_t + \epsilon_t | Y_{t-1}) = P_t + \sigma_\epsilon^2.
\]

\noindent Thus

\[
\frac{P_t}{P_t + \sigma_\epsilon^2}
\]

\noindent represents how much of our prediction error is likely due to changes in the underlying state. This is the scale factor on the prediction error. It says that because a part of every observation is due to noise, we don't take every prediciton error 100\% seriously. Instead, we adjust our expectation of the state according to how much we think a prediction deviated from an observation because the state unexpectedly changed, rather than because contemporaneous noise was especially high. How much we anticipate state to unexpectedly change is $P_t$, and we update scale by

\[
\frac{\sigma_\epsilon^2}{P_t + \sigma_\epsilon^2},
\]

\noindent the share of noise in prediction error variance. Note that this always increases our certainty about the state, which makes sense given we are adding new information to the model. We may not increase our certainty very much, however, if our observations and hence prediction errors are very noisey.

$P_t$ is calculated with each observation as a part of the one-step ahead forecast process, and $\sigma_\epsilon^2$ is assumed to be known. This forecast process appears to be a part of both fitting the model as well as forecasting itself.

To make our one-step ahead forecast for the state of our system, we go back to our state model,

\[
a_{t+1} = \mathbb{E}(\alpha_t + \eta_t | Y_t) = \mathbb{E}(\alpha_t | Y_t) = a_{t|t} = a_t + \frac{P_t v_t}{P_t + \sigma_\epsilon^2}
\]

\noindent and

\[
P_{t+1} = Var(\alpha_t + \eta_t | Y_t) = P_{t|t} + \sigma_\eta^2 = \frac{P_t \sigma_\epsilon^2}{P_t + \sigma_\epsilon^2} + \sigma_\eta^2.
\]

\noindent Because we have a random walk, our best forecast of the state's mean is our most up-to-date expectation, and the variance of that forecast is our most up-to-date variance of the state plus our current understanding of how that state's random walk evolves.

Perhaps unsurprisingly, one of the scaling factors introduced above plays a special role in filtering, 

\begin{align*}
F_t &= Var(v_t | Y_{t-1}) = P_t + \sigma_\epsilon^2, & K_t &= \frac{P_t}{F_t}.
\end{align*}

\noindent The \textbf{Kalman gain} $K_t$ is the scaling factor on the prediction errors. We can succintly write the rules for updating from $t$ to $t+1$ as

\begin{align*}
v_t &= y_t - a_t, & F_t &= P_t + \sigma_\epsilon^2, \\
a_{t|t} &= a_t + K_t v_t, & P_{t|t} &= P_t(1-K_t), \\
a_{t+1} &= a_t + K_t v_t, & P_{t+1} &= P_t (1-K_t) + \sigma_\eta^2.
\end{align*}

\subsection{Forecast errors}

The forecast errors $v_t$ and their variance $F_t$ are used in a variety of ways in state space modeling because they represent the new information we learn about $y_t$ that could not be predicted given our past observations. The forecast errors are often referred to as innovations because they represent new information in this way.

The forecast errors are mutually independent. To see this, start with the joint probability of a set of observations and use the definition of a marginal distribution

\[
p(y_1, ..., y_n) = p(y_1) \prod_{t = 2}^n p(y_t | Y_{t-1}).
\]

\noindent To understand what happens from here, we need to understand how to apply a change of variables.


\begin{aside}[Change of Variable]
    Let's start with the one dimensional case. If we have a random variable $X$ with density function $f_X(x)$ and we wish to find the density of another random variable $Y = g(X)$, denoted $f_Y(y)$, and if $g$ is strictly monotonic, then we can find it with

    \[
    f_Y(y) = f_X(g^{-1}(y)) \left| \frac{d}{dy} g^{-1}(y) \right|.
    \]

    The absolute value is a scaling factor derived from the fact that, given the relationship $Y = g(X)$, the probability that $Y$ lies in a certain image must equal the probability that $X$ lies in the associated pre-image (note the proof of the transition from the first to second line requires knowledge of Real Analysis):

    \begin{align*}
    \mathbb{P}[Y \in [y, y+dy]] &= \mathbb{P}[X\in[x, x+dx]] \\
    f_Y(y)|dy| &= f_X(x)|dx| \\
    f_Y(y) &= f_X(x) \left| \frac{dx}{dy} \right| \\
    f_Y(y) &= f_X(g^{-1}(y)) \left| \frac{d}{dy} g^{-1}(y) \right| 
    \end{align*}

    In the multivariate (i.e. vector) case we have

    \[
    f_{\bm{Y}}(\bm{y}) = f_{\bm{X}}(g^{-1}(\bm{y})) \left| \text{det}(J_{g^{-1}}(\bm{y}))\right|
    \]
\end{aside}

Let $v_t = g(y_t) = y_t - a_t$ so $y_t = g^{-1}(v_t) = v_t + a_t$ and $\frac{d g^{-1}(v_t)}{d v_t} = 1$, and letting $\bm{y}_t$ and $\bm{v}_t$ denote the random variables of which $y_t$ and $v_t$ are observations, we can then write 

\begin{align*}
p(y_1, ..., y_n) &= f_{\bm{y}_1}(y_1) \prod_{t=2}^n f_{\bm{y}_t} (y_t | Y_{t-1}) \\
&= f_{\bm{y}_1}(v_1 + a_1) \prod_{t=2}^n f_{\bm{y}_t} (v_t + a_t | Y_{t-1}) \\
&= f_{\bm{v}_1}(v_1) \prod_{t=2}^n f_{\bm{v}_t} (v_t| Y_{t-1})
\end{align*}

The second equality holds simply because $y_t = v_t + a_t$; we have not truly changed variables yet because the underlying distribution is still that of $y_t$. In the third equality, we change to the variable to $v_t$. This has an especially simply form

\begin{align*}
f_{\bm{v}_1}(v_1) &= f_{\bm{y}_1} (v_1 + a_1),  \\
f_{\bm{v}_t \mid \bm{Y}_{t-1}}(v_t \mid Y_{t-1}) &= f_{\bm{y}_t \mid \bm{Y}_{t-1}} (v_t + a_t \mid Y_{t-1}) \text{ when } t>1,
\end{align*}

\noindent because $y_t$ is just a shift of $v_t$, and the shift $a_t$ is known conditional on $Y_{t-1}$. It may seem weird to claim that two distributions that are shifts of eachother are equal. Bear in mind, however, that the functions in the equations above have become functions of $v_t$, so it is not that we are somehow claiming that $f_{\bm{v}_t \sim \bm{Y}_{t-1}}$ perfetly overlaps $f_{\bm{y}_t \sim \bm{Y}_{t-1}}$; we are saying that the probability density of $\bm{v}_t$ taking on a certain value can be equivalently expressed in terms of either $f_{\bm{v}_t \sim \bm{Y}_{t-1}}$ or $f_{\bm{y}_t \sim \bm{Y}_{t-1}}$ when the input to the latter is shifted appropriately; this is simply the core result of the change of variable theorem, applied here.

The last thing to do is to show that

\[
f_{\bm{v}_t \mid \bm{Y}_{t-1}}(v_t \mid Y_{t-1}) = f_{\bm{v}_t}(v_t)
\]

We already know that $v_t \sim N(0, F_t)$, so any conditional dependence on $Y_{t-1}$ must come through $F_t$. But 

\[
F_t = P_t + \sigma_\epsilon^2
\]

\noindent and 

\[
P_{t+1} = \frac{P_t \sigma_\epsilon^2}{P_t + \sigma_\epsilon^2} + \sigma_\eta^2
\]

\noindent where $\sigma_\epsilon^2$, $\sigma_\eta^2$, and $P_1$ are known, so the evolution of $F_t$ is \textit{deterministic} because the evolution of $P_t$ is deterministic. A deterministic process does not depend on any random variable, so

\[
f_{\bm{v}_t \mid \bm{Y}_{t-1}}(v_t \mid Y_{t-1}) = f_{\bm{v}_t}(v_t).
\]

\noindent This means that we can finally write

\begin{align*}
p(y_1, ..., y_n) &= f_{\bm{v}_1}(v_1) \prod_{t=2}^n f_{\bm{v}_t} (v_t| Y_{t-1}) \\
&= \prod_{t=1}^n f_{\bm{v}_t} (v_t) \\
&= p(v_1, ..., v_n)
\end{align*}

\noindent and we have shown that the prediction errors are mutually independent (we could technically write that last line earlier, but I wrote it here to cement the point).

We now show that the forecast errors can be computed by a Cholesky decomposition. To recap, our  updating rules are 

\begin{align*}
v_t &= y_t - a_t, & F_t &= P_t + \sigma_\epsilon^2, \\
a_{t|t} &= a_t + K_t v_t, & P_{t|t} &= P_t(1-K_t), \\
a_{t+1} &= a_t + K_t v_t, & P_{t+1} &= P_t (1-K_t) + \sigma_\eta^2, \\
& & K_t &= \frac{P_t}{F_t}.
\end{align*}

\noindent We can recursively solve for the forecast errors given $a_1$ and the observations $Y_n$:

\begin{align*}
v_1 &= y_1 - a_1 \\
v_2 &= y_2 - a_2 = y_2 - (a_1 + K_1 v_1) &&= y_2 - a_1 - K_1 (y_1 - a_1) \\
v_3 &= y_3 - a_3 = y_3 - (a_2 + K_2 v_2) &&= y_3 - a_2 - K_2 (y_2 - a_1 - K_1 (y_1 - a_1)) \\
& &&= y_3 - (a_1 + K_1 v_1) - K_2 (y_2 - a_1) + K_2 K_1 (y_1 - a_1) \\
& &&= y_3 - a_1 - K_1 (y_1 - a_1) - K_2 (y_2 - a_1) + K_2 K_1 (y_1 - a_1) \\
& &&= y_3 - a_1 - K_2 (y_2 - a_1) - K_1 (y_1 - a_1) + K_2 K_1 (y_1 - a_1) \\
& &&= y_3 - a_1 - K_2 (y_2 - a_1) - K_1 (y_1 - a_1) (1 - K_2) \\
& &&= y_3 - a_1 - K_2 (y_2 - a_1) - K_1(1 - K_2)(y_1 - a_1)
\end{align*}

\noindent We can represent this in matrix form,

\[
v = C(Y_n - 1a_1), \text{ with } v = (v_1 \cdots v_n)'
\]

The critical part of the above is the $n \times n$ matrix C, whose structure we inductively describe above,

\begin{align*}
C &= \begin{pmatrix}
1 & 0 & 0 & & 0 \\
c_{21} & 1 & 0 && 0 \\
c_{31} & c_{32} & 1 && 0 \\
 & & & \ddots & \vdots \\
c_{n1} & c_{n2} & c_{n3} & \cdots & 1
\end{pmatrix}, \\
&\left. 
    \begin{aligned}
    c_{i, i-1} &= -K_{i-1}, \\
    c_{ij} &= -(1 - K_{i-1})(1 - K_{i-2}) \cdots (1 - K_{j+1}) K_j
    \end{aligned}
\right\} \quad \text{ for } \begin{aligned}
i&>j \\
i &=2, ...., n \\
j &= 1, ..., i-2
\end{aligned}
\end{align*}

The indexes of this last line are a bit hard to understand, so let's apply them to the closed form of $v_3$ we derived above,

\[
v_3 = y_3 - a_1 - K_2 (y_2 - a_1) - \underbrace{K_1(1 - K_2)(y_1 - a_1)}_{c_{31}}.
\]

\noindent Since we are always in the lower triangular half of the matrix, we essentially create the coefficient on $y_1 - a_1$ by factors from right to left, ``counting'' from $j$ to $i - 1$. We can then state

\[
v \sim N(0, C\Omega C')
\]

\noindent where $\Omega = \text{Var}(Y_n)$ as define by the autocovariance structure of the local level model we have been working with.

Now, we also know that $\text{Var}(v_t) = F_t$ and $\text{Cov}(v_t, v_s) = 0$ for $s\neq t$, so we can also write

\[
v \sim N(0, F)
\]

\noindent where $F$ is a diagonal matrix whose non-zero elements are $F_1, ..., F_n$. So, we can say

\[
F = C \Omega C',
\]

\noindent which is exactly the form of a Cholesky decomposition: the representation of a positive semidefinite matrix $\Omega$ as a diagonal matrix $F$. It is also userful to notice that $\Omega^{-1} = C' F^{-1} C$, which should be familiar with Cholesky decomposition for OLS estimation.

Next we define the state estimation error,

\[
x_t = \alpha_t - a_t.
\]

Apparently $\text{Var}(x_t) = P_t$, but I don't see how this is is true unconditionally, so let's try and derive it.

\begin{align*}
\text{Var}(x_t) &= \text{Var}(\alpha_t - a_t) = \text{Var}(\alpha_t) + \text{Var}(a_t) \\
&= \mathbb{E}(\alpha_t^2) - \mathbb{E}(\alpha_t)^2 + \mathbb{E}(a_t^2) - \mathbb{E}(a_t)^2 \\
&= \mathbb{E}(\mathbb{E}(\alpha_t^2 \mid Y_{t-1})) - \mathbb{E}(\mathbb{E}(\alpha_t \mid Y_{t-1}))^2 + \mathbb{E}(\mathbb{E}(a_t^2 \mid Y_{t-1})) - \mathbb{E}(\mathbb{E}(a_t \mid Y_{t-1}))^2 \\
&= \mathbb{E}(\mathbb{E}(\alpha_t^2 \mid Y_{t-1}) - \mathbb{E}(\alpha_t \mid Y_{t-1})^2) \\
&= \mathbb{E}(\text{Var}(\alpha_t \mid Y_{t-1})) \\
&= \mathbb{E}(P_t) \\
&= P_t
\end{align*}

\noindent Again, the last line holds because $P_t$ is a deterministic process given that $P_1$, $\sigma_\epsilon^2$, and $\sigma_\eta^2$ are known. The fourth line holds because as of $t-1$, $a_t$ is known. The use of the law of iterated expectations seems a little bit like magic: how can we go from being interested in the unconditional variance of our state $\alpha_t$ and our state forecast $a_t$ to being satisfied with a statement that depends on knowledge as of $t-1$? Unconditionally, our state and forecast could be \textit{anything}, right? Well, I think the answer comes down to the fact that $P_t$ holds a lot of information. The error variances $\sigma_\epsilon^2$ and $\sigma_\eta^2$ that dictate its evolution are known \textit{because} we know what our model is and observations are (I'm still now sure how these are known given those, but that is what the book states). So, while $P_t$ is defined conditionally ($\alpha_t \mid Y_{t-1} \sim N(a_t, P_t)$), its true value is \textit{not conditional}. This is why we can know what the unconditional variation in $\alpha_t$ and $x_t$ by conditioning on $Y_{t-1}$ in an application of the law of iterated expectations.

Given the definition of $x_t$, we can express our forecast error in terms of our state estimation error and state update rules for our errors between $t$ and $t+1$ that are analogous to the update rules for our model,

\begin{align*}
v_t &= y_t - a_t = \alpha_t + \epsilon_t - a_t = x_t + \epsilon_t \\
x_{t+1} &= \alpha_{t+1} - a_{t+1} \\
&= \alpha_t + \eta_t - a_t - K_t v_t \\
&= x_t + \eta_t - K_t(x_t + \epsilon_t) \\
&= L_t x_t + \eta_t - K_t \epsilon_t.
\end{align*}

\noindent where $L_t = 1- K_t$. So, our state error next period is the state error of this period attenuated by the share of the variance of noise in our observations in total variance of our observations, $1-K_t = \sigma_\epsilon^2/(P_t + \sigma_\epsilon^2)$, adjusting for the random walk of our state and the noise in our observation this period. Note that the random walk of our state directly affects state esetimation error next period, becauase it is unpredictable, while the noise in the observation this period affects it via the Kalman gain, $K_t$; this is because the Kalman gain describes how much of the observed variance is due to variation in our state as opposed to noise in the observations, so said noise's affect on state estimation error is reduced in proportion to it: only part of the noise this period is allowed to affect our state estimation, so only a part of it should affect our state estimation error.

\subsection{State Smoothing}

So, how do we estimate $\alpha_1, ..., \alpha_n$ given our sample $Y_n$? We can state the following

\begin{align*}
    \alpha_t \mid Y_n &\sim N(\hat{\alpha}_t, V_t), \\
    \hat{\alpha}_t &= \mathbb{E}(\alpha_t \mid Y_n), &&\text{the smoothed state}, \\
    V_t &= \text{Var}(\alpha_t \mid Y_n), && \text{smoothed state variance}.
\end{align*}

\noindent Also, remember the following results proved above

\begin{enumerate}
    \item All forecast errors $v_1, ..., v_n$ are mutually independent.
    \item The ``future'' forecast errors $v_t, ..., v_n$ are independent of past observations $y_1, ..., y_{t-1}$ and have zero mean.
    \item $Y_n$ is fixed if and only if $Y_{t-1}$ and $v_t, ..., v_n$ are fixed.
\end{enumerate}

%Why can't we say all forecast errors are zero mean?

\noindent Given this we can state

\[
\hat{\alpha}_t = \mathbb{E}(\alpha_t \mid Y_n) = \mathbb{E}(\alpha_t \mid Y_{t-1}, v_t, ..., v_n)
\]

\noindent And note that since $v_t = y_t - a_t = y_t - \mathbb{E}(y_t | Y_{t-1})$, the forecast errors from $t$ on and $Y_{t-1}$ are independent. And since they are independent, the expected value of the state depends on a linear combination of the individual relationships between the state and those variables:

\[
\hat{\alpha}_t = \mathbb{E}(\alpha_t \mid Y_{t-1}) + \sum_{j=t}^n \dfrac{\text{Cov}(\alpha_t, v_j)}{\text{Var}(v_j)} (v_j - \mathbb{E}(v_j)) = a_t + \sum_{j=t}^n \dfrac{\text{Cov}(\alpha_t, v_j)}{F_j} v_j
\]

So adding in the rest of the sample adjusts our estimation of the state at time $t$ according to how it changes with future forecast errors: we take our understanding of what happens throughout the sample to refine what the state must have been at any point during the sample.

So, we also know we can express our forecast errors in terms of state estimation errors $x_t$, $v_t = x_t + \epsilon_t$. So, we can write these covariances compactly,

\begin{align*}
\text{Cov}(\alpha_t, v_{t+j}) = \text{Cov}(a_t + x_t, v_{t+j}) &= \text{Cov}(x_t, v_{t+j}) \\
&= P_t L_t \cdots L_{t+j}
\end{align*}

This result depends on the form of our model, I think (but i'm not sure). Remember, $L_t = 1 - K_t$. So, we can write our smoothed state as

\begin{align*}
\hat{\alpha}_t &= a_t + P_t r_{t-1} \\
r_{t-1} &= \frac{v_t}{F_t} + L_t \frac{v_{t+1}}{F_{t+1}} + \cdots + L_t \cdots L_{n-1} \frac{v_n}{F_n}
\end{align*}

\noindent $r_{t-1}$ is a weighted sum of innovations (forecast errors) after $t-1$. Using this, we can calculate the smoothed state with backwards recursion,

\begin{align*}
r_{t-1} &= \frac{v_t}{F_t} + L_t r_t \\
\hat{\alpha}_t &= a_t + P_t r_{t-1} \\
r_n &= 0
\end{align*}

Next we will calculate the \textbf{smoothed state variance}, $V_t = \text{Var}(\alpha_t \mid Y_n)$. Again, since $Y_{t-1}$ and the $v_t$s are independent, 

\[
V_t = \text{Var}(\alpha_t | Y_{t-1}, v_1, ..., v_n) = P_t - \sum_{j=t}^n \frac{\text{Cov}(\alpha_t, v_j)^2}{F_j}.
\]

\noindent Note the minus sign due to the addition of information. We use the same relation between $x_t$ and $v_t$ to express the covariances above,

\begin{align*}
V_t &= P_t - P_t^2 N_{t-1} \\
N_{t-1} &= \frac{1}{F_t} + \frac{L_t^2}{F_{t+1}} + \cdots + \frac{L_t^2 \cdots L_{n-1}^2}{F_n} \\
N_n &= 0
\end{align*}

So we can solve for the smoothed state variances through backwards recursion as well,

\begin{align*}
N_{t-1} &= \frac{1}{F_t} + L_t^2 N_t \\
V_t &= P_t - P_t^2 N_{t-1} 
\end{align*}

How do we \textbf{initialize} the Kalman filter, given that we assume that $a_1$ and $P_1$ are known? Start with a diffuse prior: give $a_1$ some arbitrary value and let $P_1 \rightarrow \infty$. We know our model starts out as

\begin{align*}
v_1 &= y_1 - a_1, & F_1 &=P_1 + \sigma_\epsilon^2
\end{align*}

\noindent so given our assumptions we can say

\begin{align*}
a_2 &= a_1 + \frac{P_1}{P_1 + \sigma_\epsilon^2}(y_1 - a_1) &&= y_1 \\
P_2 &= P_1 \left( 1 - \frac{P_1}{P_1 + \sigma_\epsilon^2}\right) + \sigma_\eta^2 &&= \sigma_\epsilon^2 + \sigma_\eta^2
\end{align*}

Note that the Kalman filter is defined for $t \geq 2$, this also means our smoothing recursions are defined in this time window as well. To know what our smoothing recursions yield for $t=1$ given our diffuse prior, we need to do more work. From the conditional expectation of our state, we get

\begin{align*}
\hat{\alpha}_1 &= a_1 + P_1 \left[ \frac{1}{P_1 + \sigma_\epsilon^2} v_1 + \left( 1 - \frac{P_1}{P_1 + \sigma_\epsilon^2} r_1 \right) \right] \\
&= a_1 + v_1 + \sigma_\epsilon^2 r_1 \\
&= y_1 + \sigma_\epsilon^2 r_1
\end{align*}

\noindent And by a similar calculation, 

\[
V_1 = \sigma_\epsilon^2 - (\sigma_\eta^2)^2 N_1
\]



\subsection{Estimation exercise: GRAB closes}

Actual estimation is a multistep process. The main parameters we are estimating in this model are the noise variances $\sigma_\epsilon^2$ and $\sigma_\eta^2$, but the likelihood we are maximizing comes from the Kalman filter results. So maximizing the likelihood involes calculating the filter given some values of the noise variance, and then checking the log likelihood of the resulting Kalman filter. Estimation is numerically solved by iterating the assumed values of the noise variances until we converge on a maximum log likelihood.


Apply Kalman filter to get $v_t, F_t, a_t, P_t$, then apply backward recursions to get $\hat{\alpha}_t, V_t, r_t, N_t$.

\subsection{What's really the difference between conditioning on $Y_n$ vs $Y_{t-1}$}
If we note that 

\[
\hat{\alpha}_t = \mathbb{E}(\alpha_t \mid Y_n)
\]

\noindent then we can extend this to the multivariate case and apply it to our local level model:

\begin{align*}
p(\alpha_t, Y_n) &= p(Y_n) p(\alpha_t \mid Y_n) \\
f_{\alpha_t, Y_n} (\alpha_t, Y_n) &= f_{Y_n} (Y_n) f_{\alpha_t \mid Y_n} (\alpha_t \mid Y_n)  \\
\frac{f_{\alpha_t, Y_n} (\alpha_t, Y_n)}{f_{Y_n} (Y_n)} &= f_{\alpha_t \mid Y_n} (\alpha_t \mid Y_n)
\end{align*}

\noindent and we have for our local level model

\begin{align*}
Y_n &\sim N(\bm{1}a_1, \Omega) \\
\alpha_t = y_t - \epsilon_t &\sim N(a_1, \sigma_\epsilon^2 + (t-1)\sigma_\eta^2 + \sigma_\epsilon^2)
\end{align*}

\noindent To understand where this is coming from, note that our model of $y_t$ can be expressed as

\[
y_t = a_1 + \sum_{j=1}^{t-1} \eta_j + \epsilon_t
\]

\noindent because each period we have $\epsilon_t$ and $\eta_t$ but $\eta_t$ only affects the next period, more explicitly

\[
Y_n = \begin{pmatrix}
a_1 + \epsilon_1 \\
a_1 + \eta_1 + \epsilon_2 \\
a_1 + \eta_1 + \eta_2 + \epsilon_3 \\
\vdots \\
a_1 + \sum_{t=1}^{n-1} \eta_t + \epsilon_n
\end{pmatrix}
\]

\noindent and so

\begin{align*}
\Omega &= \text{Var}(Y_n) = \mathbb{E} [(Y_n - \mathbb{E}(Y_n))(Y_n - \mathbb{E})'] \\
&= \begin{pmatrix}
\text{Var}(y_1) & \text{Cov} (y_1, y_2) & \cdots & \text{Cov}(y_1, y_n) \\
\text{Cov}(y_2, y_1) & \text{Var}(y_2) & & \\
\vdots & & \ddots & \\
\text{Cov}(y_n, y_1) & & & \text{Var}(y_n)
\end{pmatrix} \\
&= \begin{pmatrix}
P_1 + \sigma_\epsilon^2 & P_1 & P_1 & \cdots & P_1 \\
P_1 & P_1 + \sigma_\eta^2 + \sigma_\epsilon^2 & P_1 + \sigma_\eta^2 & & \\
P_1 & P_1 + \sigma_\eta^2 & P_1 + 2 \sigma_\eta^2 + \sigma_\epsilon^2 & & \\
\vdots & & & \ddots & \\
P_1 & & & & P_1 + (n-1) \sigma_\eta^2 + \sigma_\epsilon^2
\end{pmatrix}
\end{align*}

\noindent Note how the variances due to shocks to the random walk enter the elements and where they are absent. They enter when two observations share historical shocks: $\sigma_\eta^2$ is present after the first period, in proportion to the overlap in historical periods (i.e. in proportion to the earlier observations shocks). Also note where variance to the idiosyncratic shocks $\epsilon_t$ appears: only on diagonal elements.

\[
\begin{pmatrix}
\alpha_t \\
Y_n
\end{pmatrix} \sim N \left( a_t \bm{1}_{n+1}, \begin{pmatrix}
\text{Var}(\alpha_t) & \text{Cov}(\alpha_t, Y_n') \\
\text{Cov}(Y_n, \alpha_t) & \text{Var}(Y_n)
\end{pmatrix} \right)
\]

\noindent where

\[
\begin{pmatrix}
\text{Var}(\alpha_t) & \text{Cov}(\alpha_t, Y_n') \\
\text{Cov}(Y_n, \alpha_t) & \text{Var}(Y_n)
\end{pmatrix} = \begin{pmatrix}
\text{Var}(y_t - \epsilon_t) & \text{Cov}(y_t - \epsilon_t, y_1) & \cdots & \text{Cov}(y_t - \epsilon_t, y_t) & \cdots & \text{Cov}(y_t - \epsilon_t, y_n) \\
\text{Cov}(y_1, y_t - \epsilon_t) & & & & \\
\vdots & & & & \\
\text{Cov}(y_t, y_t - \epsilon_t) & & & \Omega & \\
\vdots & & & & \\
\text{Cov}(y_n, y_t - \epsilon_t) & & & & \\
\end{pmatrix}
\]

\noindent We are not adding redundant data to the vector $Y_n$ because we are adding the state explicitly, which requires considering the difference between the observation and its error as opposed to just the observation, therefore adding new information to the system. The multivariate normal distribution is

\[
f_{\bm{X}} (x_1, ..., x_n) = \frac{1}{\sqrt{(2\pi)^n |\bm{\Sigma}|}} \text{exp} \left(\frac{-1}{2} (\bm{x} - \bm{\mu_x})' \bm{\Sigma}^{-1} (\bm{x} - \bm{\mu_x}) \right) = N(\bm{\mu_x}, \bm{\Sigma})
\]

\noindent so

\[
f_{\alpha_t, Y_n}(\alpha_t, Y_n) = \text{const.} \times \exp \left( \dfrac{-1}{2} \left( \begin{pmatrix}
\alpha_t \\
Y_n
\end{pmatrix} - a_1 \bm{1}_{n+1}\right)' \begin{pmatrix}
\text{Var}(\alpha_t) & \text{Cov}(\alpha_t, Y_n') \\
\text{Cov}(Y_n, \alpha_t) & \Omega
\end{pmatrix}^{-1} \left( \begin{pmatrix}
\alpha_t \\
Y_n
\end{pmatrix} - a_1 \bm{1}_{n+1} \right)\right)
\]

\noindent and

\[
f_{Y_n}(Y_n) = \text{const.} \times \exp \left( 
    \dfrac{-1}{2} \left( Y_n - a_1 \bm{1}_n \right)' \Omega^{-1} \left( Y_n - a_1 \bm{1}_n \right)
\right)
\]

\noindent All the action is taking place in the exponential, so, dropping the $-1/2$,

\begin{align*}
\text{exponential} &= \left( \begin{pmatrix}
\alpha_t \\
Y_n
\end{pmatrix} - a_1 \bm{1}_{n+1}\right)' \begin{pmatrix}
\text{Var}(\alpha_t) & \text{Cov}(\alpha_t, Y_n') \\
\text{Cov}(Y_n, \alpha_t) & \Omega
\end{pmatrix}^{-1} \left( \begin{pmatrix}
\alpha_t \\
Y_n
\end{pmatrix} - a_1 \bm{1}_{n+1} \right) \\
& - \left( Y_n - a_1 \bm{1}_n \right)' \Omega^{-1} \left( Y_n - a_1 \bm{1}_n \right)
\end{align*}

\noindent We want to isolate what equates to the mean of $\alpha_t$. Consequently, we don't really care about the second term being substracted, because it will only affect terms in the final expression that haveelements of $Y_n$, not $\alpha_t$.
